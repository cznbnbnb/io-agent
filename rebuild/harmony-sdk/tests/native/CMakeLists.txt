cmake_minimum_required(VERSION 3.16)
project(turboio_native_regression LANGUAGES C CXX)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../../core-probe/Vendor/opus-1.5.2 ${CMAKE_CURRENT_BINARY_DIR}/opus)
add_executable(audio-environment-native ../audio-environment-native.cpp)
add_executable(recording-native ../recording-native.cpp)
foreach(target audio-environment-native recording-native)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE ../native-stubs)
  target_link_libraries(${target} PRIVATE opus)
endforeach()
enable_testing()
add_test(NAME audio-environment-native COMMAND audio-environment-native)
add_test(NAME recording-native COMMAND recording-native)
